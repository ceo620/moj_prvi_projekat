#!/usr/bin/env python3

import json
from pathlib import Path


def build_markdown(report):

    lines = [
        "# FINAL VALIDATION REPORT",
        "",
        "- Module: " + str(report["MODULE"]),
        "- Total modules: " + str(report["TOTAL_MODULES"]),
        "- Total files: " + str(report["TOTAL_FILES"]),
        "- Total errors: " + str(report["TOTAL_ERRORS"]),
        "",
        "## Validator Summary",
        "",
        "| Validator | Files | Errors |",
        "|---|---:|---:|",
    ]

    for module in report["REPORTS"]:

        errors = len([
            item
            for item in module["RESULTS"]
            if item["STATUS"] != "PASS"
        ])

        lines.append(
            "| {} | {} | {} |".format(
                module["MODULE"],
                module["TOTAL_RESULTS"],
                errors
            )
        )

    lines.append("")
    lines.append("## Framework Status")
    lines.append("")
    lines.append(
        "PASS"
        if report["TOTAL_ERRORS"] == 0
        else "FAIL"
    )

    return "\n".join(lines) + "\n"


def main():

    input_path = Path(
        "reports/FINAL_VALIDATION_REPORT.json"
    )

    output_path = Path(
        "reports/FINAL_VALIDATION_REPORT.md"
    )

    report = json.loads(
        input_path.read_text(
            encoding="utf-8"
        )
    )

    output_path.write_text(
        build_markdown(report),
        encoding="utf-8"
    )

    print("MODULE=MARKDOWN_REPORT_ENGINE")
    print("OUTPUT=" + str(output_path))
    print("STATUS=PASS")

    return 0


if __name__ == "__main__":
    raise SystemExit(main())
