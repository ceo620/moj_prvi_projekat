#!/usr/bin/env python3

import json
from pathlib import Path


SCHEMA = (
    "https://docs.oasis-open.org/sarif/sarif/"
    "v2.1.0/cos02/schemas/sarif-schema-2.1.0.json"
)


def make_result(module, item):

    file_name = item.get("FILE", "unknown")

    message = item.get(
        "ERROR",
        module + " reported a validation finding"
    )

    return {
        "ruleId": module,
        "level": "error",
        "message": {
            "text": str(message)
        },
        "locations": [
            {
                "physicalLocation": {
                    "artifactLocation": {
                        "uri": str(file_name)
                    }
                }
            }
        ]
    }


def build_sarif(report):

    results = []

    rules = []

    for module in report["REPORTS"]:

        module_name = module["MODULE"]

        rules.append({
            "id": module_name,
            "name": module_name,
            "shortDescription": {
                "text": module_name.replace("_", " ").title()
            }
        })

        for item in module["RESULTS"]:

            if item.get("STATUS") != "PASS":
                results.append(
                    make_result(module_name, item)
                )

    return {
        "$schema": SCHEMA,
        "version": "2.1.0",
        "runs": [
            {
                "tool": {
                    "driver": {
                        "name": "FREYA 40_VALIDATORS",
                        "version": "1.0.0",
                        "informationUri": (
                            "https://docs.oasis-open.org/sarif/"
                        ),
                        "rules": rules
                    }
                },
                "results": results
            }
        ]
    }


def main():

    source = Path(
        "reports/FINAL_VALIDATION_REPORT.json"
    )

    output = Path(
        "reports/FINAL_VALIDATION_REPORT.sarif"
    )

    report = json.loads(
        source.read_text(encoding="utf-8")
    )

    sarif = build_sarif(report)

    output.write_text(
        json.dumps(
            sarif,
            indent=2,
            ensure_ascii=False
        ),
        encoding="utf-8"
    )

    result_count = len(
        sarif["runs"][0]["results"]
    )

    print("MODULE=SARIF_EXPORT_ENGINE")
    print("VERSION=2.1.0")
    print("RESULT_COUNT=" + str(result_count))
    print("OUTPUT=" + str(output))
    print("STATUS=PASS")

    return 0


if __name__ == "__main__":
    raise SystemExit(main())
