from pathlib import Path
import os,json,ast,re,time,hashlib,tempfile,fcntl,stat
d=Path(__file__).resolve().parent
state=d/"LARGE_SCAN_STATE.json"
lock=(d/"LOCK").open("a")
fcntl.flock(lock,fcntl.LOCK_EX|fcntl.LOCK_NB)
j=json.loads(state.read_text())
assert j["version"]==1 and len(j["files"])==2
cfg=json.loads((d/"SCOPE.json").read_text())
raw=(d/"harvest.original.py").read_bytes()
h=hashlib.sha256(raw).hexdigest()
assert h==cfg["source_sha256"]
assert j.get("terms_sha",h)==h
j["terms_sha"]=h
SIG=next(ast.literal_eval(n.value) for n in ast.parse(raw).body
    if isinstance(n,ast.Assign) and any(
    isinstance(t,ast.Name) and t.id=="SIG" for t in n.targets))
E=re.compile(r'(?<![\w.+-])[\w.+-]+@[\w.-]+\.[A-Za-z]{2,}')
def meta(s):
    assert stat.S_ISREG(s.st_mode)
    return [s.st_dev,s.st_ino,s.st_size,s.st_mtime_ns,s.st_ctime_ns]

def save():
    fd,tmp=tempfile.mkstemp(dir=d)
    try:
        with os.fdopen(fd,"w") as f:
            json.dump(j,f,ensure_ascii=False)
            f.flush()
            os.fsync(f.fileno())
        os.replace(tmp,state)
    finally:
        if os.path.exists(tmp): os.unlink(tmp)

started=time.monotonic()
count=0
while count<16 and time.monotonic()-started<15:
    x=next((v for v in j["files"]
        if v["offset"]<v["meta"][2]),None)
    if x is None: break
    p=Path(x["path"])
    assert meta(p.lstat())==x["meta"], "HOLD=SOURCE_CHANGED"
    offset=x["offset"]
    with p.open("rb") as f:
        assert meta(os.fstat(f.fileno()))==x["meta"]
        f.seek(offset)
        data=f.read(1048576)
        if data and not data.endswith(b"\n"):
            if f.tell()<x["meta"][2]:
                data+=f.readline(262145)
        end=f.tell()
        assert data and len(data)==end-offset
        assert data.endswith(b"\n") or end==x["meta"][2], "HOLD=LONG_LINE"
        assert meta(os.fstat(f.fileno()))==x["meta"]
    assert meta(p.lstat())==x["meta"]
    assert b"\0" not in data, "HOLD=NUL_BYTES"
    text=data.decode("utf-8")
    contacts=set(x["contacts"])
    parts=text.split("@")
    for left,right in zip(parts,parts[1:]):
        a=re.match(r"[\w.+-]+",left[::-1])
        b=re.match(r"[\w.-]+",right)
        if a and b:
            m=E.match(a[0][::-1]+"@"+b[0])
            if m: contacts.add(m[0])
    signals=set(x["signals"])
    low=text.lower() if len(signals)<len(SIG) else ""
    signals.update(k for k,words in SIG.items()
        if k not in signals and any(w in low for w in words))
    assert meta(p.lstat())==x["meta"], "HOLD=SOURCE_CHANGED"
    x["blocks"].append([offset,len(data),hashlib.sha256(data).hexdigest()])
    x.update(offset=end,contacts=sorted(contacts),signals=sorted(signals))
    save()
    count+=1

print("BLOCKS_THIS_BATCH="+str(count))
for i,x in enumerate(j["files"],1):
    print("FILE_"+str(i)+"_BYTES="+str(x["offset"])+"/"+str(x["meta"][2]))
print("READ_COMPLETE="+str(all(x["offset"]==x["meta"][2] for x in j["files"])))
print("MAIN_HOLDS=UNCHANGED\nEXTERNAL_SEND=NONE")
lock.close()
