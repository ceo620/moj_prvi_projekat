#!/usr/bin/env python3
import hashlib,json,os,sys,tempfile
def hold(r,c):
 print("RESULT=HUMAN_HOLD");print("REASON="+r);raise SystemExit(c)
if len(sys.argv)!=3: hold("USAGE",64)
src,outdir=sys.argv[1:]
if not os.path.isfile(src): hold("DOCUMENT_NOT_FOUND",65)
h=hashlib.sha256()
with open(src,"rb") as f:
 for b in iter(lambda:f.read(65536),b""): h.update(b)
digest=h.hexdigest()
with open(src,"rb") as f: data=f.read()
try: text=data.decode("utf-8")
except UnicodeDecodeError: hold("DOCUMENT_ENCODING_UNSUPPORTED",66)
meta={"object_id":digest,"source_name":os.path.basename(src),"size_bytes":len(data),"line_count":len(text.splitlines()),"word_count":len(text.split()),"status":"PROCESSED"}
os.makedirs(outdir,exist_ok=True)
target=os.path.join(outdir,digest+".json")
fd,tmp=tempfile.mkstemp(prefix=".document.",dir=outdir)
with os.fdopen(fd,"w",encoding="utf-8") as f:
 json.dump(meta,f,sort_keys=True,separators=(",",":"));f.write("\n");f.flush();os.fsync(f.fileno())
os.replace(tmp,target)
print("OBJECT_ID="+digest)
print("OUTPUT_FILE="+target)
print("WORD_COUNT="+str(meta["word_count"]))
print("LINE_COUNT="+str(meta["line_count"]))
print("RESULT=PASS")
