import os,sys,hashlib,datetime

ROOT="/mnt"

G="\033[1;32m"; C="\033[1;36m"; Y="\033[1;33m"
R="\033[1;31m"; W="\033[1;37m"; X="\033[0m"

def cls():
    print("\033[2J\033[H",end="")

def walk_files(depth=4):
    for base,dirs,files in os.walk(ROOT):
        d=base[len(ROOT):].count(os.sep)
        if d>=depth:
            dirs[:]=[]
        for f in files:
            yield os.path.join(base,f)

def stats():
    data={"FILES":0,"PDF":0,"DOCX":0,"XLSX":0,"CSV":0,"ARCHIVES":0}
    for p in walk_files():
        data["FILES"]+=1
        q=p.lower()
        if q.endswith(".pdf"): data["PDF"]+=1
        if q.endswith(".docx"): data["DOCX"]+=1
        if q.endswith(".xlsx"): data["XLSX"]+=1
        if q.endswith(".csv"): data["CSV"]+=1
        if q.endswith((".zip",".tar",".tar.gz",".tgz",".gz",".bz2",".xz",".7z",".rar")):
            data["ARCHIVES"]+=1
    return data

def recent():
    header()
    a=[]
    for p in walk_files():
        try: a.append((os.path.getmtime(p),p))
        except: pass
    a.sort(reverse=True)
    print(W+"20 MOST RECENT FILES\n"+X)
    for t,p in a[:20]:
        dt=datetime.datetime.fromtimestamp(t).strftime("%Y-%m-%d %H:%M")
        print(dt+"  "+p[len(ROOT)+1:])
    input("\nReturn ↵ ")

def docs():

def archives():
    header()
    a=[p for p in walk_files(6) if p.lower().endswith((".zip",".tar",".tar.gz",".tgz",".gz",".bz2",".xz",".7z",".rar"))]
    print(W+"ARCHIVES = "+str(len(a))+"\n"+X)
    for p in a:
        try: z=os.path.getsize(p)
        except: z=0
        print("%9d  %s"%(z,p[len(ROOT)+1:]))

def integrity():
    header()
    rows=[]
    for p in walk_files(3):
        try:
            rel=p[len(ROOT)+1:]
            z=os.path.getsize(p)
            rows.append(rel+"|"+str(z))
        except: pass
    h=hashlib.sha256(("\n".join(sorted(rows))+"\n").encode()).hexdigest()
    print(W+"READ-ONLY FILE MANIFEST"+X)
    print("\nObjects:",len(rows))
    print("SHA256:")
    print(h)
    print("\n"+G+"PATH/SIZE MANIFEST ONLY"+X)
    input("\nReturn ↵ ")


def search():
    header()
    q=input("Search term > ").strip().lower()
    if not q:
        return
    hits=[]
    for p in walk_files(5):
        rel=p[len(ROOT)+1:]
        if q in rel.lower():
            hits.append(rel)
    print("\nRESULTS =",len(hits),"\n")
    for p in hits[:40]:
        print(" "+p)
    if len(hits)>40:
        print(" ...")
    input("\nReturn ↵ ")

def live_todo():
    import os,datetime
    header()
    f="/root/FREYA_IPHONE_ISH_NODE_888/15_RUNTIME/HUMAN_GATE_TODO.tsv"
    print(W+"LIVE TODO · 24H\n"+X)
    if not os.path.isfile(f):
        print("No active tasks configured.")
    else:
        now=datetime.datetime.now()
        for line in open(f):
            x=line.rstrip().split("\t")
            if len(x)<3: continue
            task,deadline,status=x[:3]
            try:
                d=datetime.datetime.fromisoformat(deadline)
                sec=(d-now).total_seconds()
                if status=="DONE": state="DONE"
                elif sec<0: state="OVERDUE"
                elif sec<=3600: state="DUE <1H"
                else: state="ACTIVE"
                print("[%s] %s"%(state,task))
                print(" Deadline:",deadline)
            except: print("[REVIEW]",task)
    input("\nReturn ↵ ")
