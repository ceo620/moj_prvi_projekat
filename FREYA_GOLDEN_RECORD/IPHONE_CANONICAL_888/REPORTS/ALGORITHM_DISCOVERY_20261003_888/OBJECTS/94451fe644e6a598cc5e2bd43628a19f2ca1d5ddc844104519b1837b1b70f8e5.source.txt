#!/usr/bin/env python3
import json, os, urllib.request, xml.etree.ElementTree as ET, html

IMENIK_PATH = "/root/faza0_skripte/vip_kontakti.json"
# Promjena na garantovano vidljivu iOS lokaciju
EMAIL_IDEAS_DIR = "/root/Documents/SOBA_4_EXPORTS_AND_OUTBOX"
os.makedirs(EMAIL_IDEAS_DIR, exist_ok=True)

with open(IMENIK_PATH, "r") as f:
    kontakti = json.load(f)

try:
    url = "https://www.europarl.europa.eu/rss/doc/top-stories/en.xml"
    req = urllib.request.Request(url, headers={'User-Agent': 'Mozilla/5.0'})
    with urllib.request.urlopen(req, timeout=10) as response:
        tema = html.unescape(ET.fromstring(response.read()).find('.//item/title').text)
except Exception:
    tema = "Global technology infrastructure shifts"

fokus = "secure Unix sandboxing and high-integrity data routing"

for osoba in kontakti:
    ime = osoba['ime']
    if "Asistent" in ime: continue

    email_adresa = osoba.get('email', f"{ime.lower().replace(' ', '.')}@domain.com")
    attachment_path = "/root/freya_audit/izlaz/FREYA_IPHONE/99_EXPORTS/FREYA_PACKET_LATEST.tar.gz"
    attachment_info = attachment_path if os.path.exists(attachment_path) else "NONE"

    if "Elon" in ime:
        subject = "Freja Core & X Infrastructure Integration"
        text = f"Dear Elon,\n\nRegarding recent shifts in '{tema}', we have stabilized a unified framework for {fokus}.\n\nLet's do a 5-minute technical sync next week to discuss direct infrastructure integration.\n\nBest,\nDanijela Djurovic Keskin\nFounder, Freja Core"
    elif "Danijela" in ime:
        subject = "Internal Archive: Core System Update"
        text = f"System Brief:\n- Market Signal: {tema}\n- Focus: {fokus}\n\nAll sandboxed modules verified."
    else:
        subject = "Partnership Proposal: Infrastructure Integration"
        text = f"Dear {ime},\n\nFollowing trends in '{tema}', our team is deploying high-performance systems for {fokus}.\n\nAre you available for a brief introductory call next week to explore integration?\n\nSincerely,\nDanijela Djurovic Keskin"

    with open(f"{EMAIL_IDEAS_DIR}/MAIL_{ime.replace(' ', '_')}.txt", "w") as mf:
        mf.write(f"KOME: {email_adresa}\nSUBJECT: {subject}\nATTACHMENT: {attachment_info}\n----------------------------------\nTEXT:\n{text}\n")
    print(f"🧠 Kreiran vidljivi mail za: {ime}")
