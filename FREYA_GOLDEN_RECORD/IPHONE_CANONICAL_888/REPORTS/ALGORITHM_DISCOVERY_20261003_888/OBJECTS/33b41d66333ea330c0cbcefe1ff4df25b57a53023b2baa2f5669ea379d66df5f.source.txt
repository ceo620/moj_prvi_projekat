#!/usr/bin/env python3
import datetime
import os
import sqlite3
import sys

BASE = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
DB = os.path.join(BASE, "07_STATE", "freya_one.db")

def now():
    return datetime.datetime.utcnow().replace(microsecond=0).isoformat() + "Z"

def connect():
    db = sqlite3.connect(DB)
    db.execute("PRAGMA journal_mode=WAL")
    db.executescript("""
    CREATE TABLE IF NOT EXISTS systems (
      name TEXT PRIMARY KEY,
      role TEXT NOT NULL,
      state TEXT NOT NULL,
      target TEXT,
      updated_utc TEXT NOT NULL
    );
    CREATE TABLE IF NOT EXISTS jobs (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      title TEXT NOT NULL,
      state TEXT NOT NULL,
      created_utc TEXT NOT NULL,
      updated_utc TEXT NOT NULL
    );
    CREATE TABLE IF NOT EXISTS events (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      event_utc TEXT NOT NULL,
      kind TEXT NOT NULL,
      message TEXT NOT NULL
    );
    """)
    return db

def event(db, kind, message):
    db.execute("INSERT INTO events(event_utc,kind,message) VALUES(?,?,?)", (now(), kind, message))

def init():
    db = connect()
    systems = [
      ("MAGNUS", "ORCHESTRATOR", "REGISTERED", "FREYA_ONE_QUEUE"),
      ("DARKO", "CONTROLLED_EXECUTOR_ADAPTER", "NOT_ACTIVATED", "V1R3_LAUNCHER"),
      ("DELTA", "VALIDATOR", "NOT_ACTIVATED", "FREYA_ONE_VALIDATION"),
      ("HUMAN_GATE", "FINAL_AUTHORITY", "ACTIVE", "DANIJELA_DJUROVIC_KESKIN"),
    ]
    for name, role, state, target in systems:
        db.execute("INSERT OR REPLACE INTO systems VALUES(?,?,?,?,?)", (name, role, state, target, now()))
    event(db, "BUILD", "FREYA_ONE_CORE_CREATED")
    db.commit()
    print("RESULT=FREYA_ONE_INITIALIZED")

def status():
    db = connect()
    print("=== FREYA ONE 888 ===")
    for state, count in db.execute("SELECT state,COUNT(*) FROM jobs GROUP BY state ORDER BY state"):
        print(f"JOBS_{state}={count}")
    total = db.execute("SELECT COUNT(*) FROM jobs").fetchone()[0]
    print(f"JOBS_TOTAL={total}")
    print("SYSTEMS=")
    for name, role, state in db.execute("SELECT name,role,state FROM systems ORDER BY name"):
        print(f"  {name}|ROLE={role}|STATE={state}")
    print("AUTO_EXECUTION=NO")
    print("HUMAN_GATE=ACTIVE")

def submit(title):
    if not title.strip():
        raise SystemExit("BLOCKER=EMPTY_JOB_TITLE")
    db = connect()
    stamp = now()
    cur = db.execute("INSERT INTO jobs(title,state,created_utc,updated_utc) VALUES(?,?,?,?)", (title.strip(), "PENDING", stamp, stamp))
    event(db, "JOB_SUBMITTED", f"JOB_ID={cur.lastrowid}")
    db.commit()
    print(f"JOB_ID={cur.lastrowid}")
    print("STATE=PENDING")
    print("EXECUTION=NO")

def transition(job_id, new_state):
    db = connect()
    row = db.execute("SELECT state FROM jobs WHERE id=?", (job_id,)).fetchone()
    if not row:
        raise SystemExit("BLOCKER=JOB_NOT_FOUND")
    allowed = {("PENDING", "APPROVED"), ("PENDING", "HOLD"), ("APPROVED", "HOLD")}
    if (row[0], new_state) not in allowed:
        raise SystemExit(f"BLOCKER=INVALID_TRANSITION_{row[0]}_TO_{new_state}")
    db.execute("UPDATE jobs SET state=?,updated_utc=? WHERE id=?", (new_state, now(), job_id))
    event(db, "HUMAN_GATE", f"JOB_ID={job_id}|STATE={new_state}")
    db.commit()
    print(f"JOB_ID={job_id}")
    print(f"STATE={new_state}")
    print("EXECUTION=NO")

def jobs():
    db = connect()
    for row in db.execute("SELECT id,state,title,created_utc,updated_utc FROM jobs ORDER BY id DESC LIMIT 50"):
        print("JOB_ID=%s|STATE=%s|TITLE=%s|CREATED=%s|UPDATED=%s" % row)

def usage():
    print("USAGE: freya status|systems|jobs|submit TITLE|approve ID|hold ID")

def main():
    args = sys.argv[1:]
    if not args:
        status()
    elif args[0] == "init":
        init()
    elif args[0] in {"status", "systems"}:
        status()
    elif args[0] == "jobs":
        jobs()
    elif args[0] == "submit" and len(args) > 1:
        submit(" ".join(args[1:]))
    elif args[0] in {"approve", "hold"} and len(args) == 2 and args[1].isdigit():
        transition(int(args[1]), "APPROVED" if args[0] == "approve" else "HOLD")
    else:
        usage()
        raise SystemExit(2)

if __name__ == "__main__":
    main()
