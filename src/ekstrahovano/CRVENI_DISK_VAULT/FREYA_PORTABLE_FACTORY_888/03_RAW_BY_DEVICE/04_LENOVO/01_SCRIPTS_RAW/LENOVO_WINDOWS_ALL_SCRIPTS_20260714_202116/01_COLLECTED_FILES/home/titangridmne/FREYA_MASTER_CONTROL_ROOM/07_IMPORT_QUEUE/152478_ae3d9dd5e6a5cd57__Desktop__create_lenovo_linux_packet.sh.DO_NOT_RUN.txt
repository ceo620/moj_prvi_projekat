#!/bin/zsh

# === MAC → LENOVO LINUX PACKAGE PREP ===
# MODE: READ_ONLY / COPY_ONLY / NO_DELETE / NO_MOVE / NO_RENAME / NO_RUNTIME

TS=$(date +"%Y%m%d_%H%M%S")
PKG="$HOME/Desktop/LENOVO_LINUX_READY_PACKET_$TS"

mkdir -p "$PKG/00_CONTROL"
mkdir -p "$PKG/01_PAYLOAD"
mkdir -p "$PKG/02_HASH_REGISTER"
mkdir -p "$PKG/03_MANIFEST"
mkdir -p "$PKG/04_HUMAN_REVIEW"

cat > "$PKG/00_CONTROL/README_FIRST.txt" <<EOF
LENOVO LINUX READY PACKET
TIMESTAMP=$TS
SOURCE_NODE=MAC
TARGET_NODE=LENOVO_LINUX
MODE=READ_ONLY
DELETE=BLOCKED
MOVE=BLOCKED
RENAME=BLOCKED
RUNTIME=BLOCKED
COPY_ONLY=YES

PURPOSE:
This packet is prepared on Mac for transfer to Lenovo Linux.
Lenovo must only inspect, hash, classify, and register.
No script execution is approved.
EOF

cat > "$PKG/03_MANIFEST/MANIFEST.txt" <<EOF
PACKET_NAME=LENOVO_LINUX_READY_PACKET_$TS
CREATED_ON_MAC=YES
TARGET=LENOVO_LINUX
HUMAN_GATE=MANDATORY
RUNTIME_APPROVED=NO
DELETE_APPROVED=NO
MOVE_APPROVED=NO
RENAME_APPROVED=NO
EOF

find "$PKG" -type f -print0 | xargs -0 shasum -a 256 > "$PKG/02_HASH_REGISTER/SHA256_REGISTER.txt"

cd "$HOME/Desktop" || exit 1
zip -r "LENOVO_LINUX_READY_PACKET_$TS.zip" "LENOVO_LINUX_READY_PACKET_$TS"

open "$HOME/Desktop"

echo "DONE"
echo "FOLDER: $PKG"
echo "ZIP: $HOME/Desktop/LENOVO_LINUX_READY_PACKET_$TS.zip"
echo "STATUS: READY_FOR_LENOVO_LINUX_TRANSFER"
