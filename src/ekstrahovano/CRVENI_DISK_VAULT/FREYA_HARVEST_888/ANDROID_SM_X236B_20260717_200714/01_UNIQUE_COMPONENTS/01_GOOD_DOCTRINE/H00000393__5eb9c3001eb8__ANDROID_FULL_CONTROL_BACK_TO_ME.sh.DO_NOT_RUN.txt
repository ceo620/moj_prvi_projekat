#!/data/data/com.termux/files/usr/bin/bash
set -u

TS="$(date +%Y%m%d_%H%M%S)"
ROOT="/sdcard/Download/ANDROID_FULL_CONTROL_BACK_TO_ME_$TS"
REPORT="$ROOT/ANDROID_FULL_CONTROL_REPORT_$TS.txt"
HASHREG="$ROOT/ANDROID_FULL_CONTROL_SHA256_REGISTER_$TS.txt"

mkdir -p "$ROOT"

log() {
  echo "$1" | tee -a "$REPORT"
}

hash_file() {
  if [ -f "$1" ]; then
    sha256sum "$1" | tee -a "$HASHREG"
  else
    echo "MISSING  $1" | tee -a "$HASHREG"
  fi
}

log "ANDROID FULL CONTROL BACK TO OPERATOR"
log "Timestamp: $(date '+%Y-%m-%d %H:%M:%S')"
log "Mode: CONTROLLED_EXPORT_ONLY"
log "Controls: NO DELETE | NO MOVE | NO LOOP | NO FAKE SYNC"
log ""

log "===== IDENTITY ====="
log "whoami: $(whoami)"
log "home: $HOME"
log "pwd: $(pwd)"
log "uname: $(uname -a)"
log ""

log "===== WIFI INFO ====="
if command -v termux-wifi-connectioninfo >/dev/null 2>&1; then
  termux-wifi-connectioninfo 2>&1 | tee -a "$REPORT"
else
  log "termux-wifi-connectioninfo not available"
fi
log ""

log "===== SSHD STATUS ====="
ps aux | grep '[s]shd' | tee -a "$REPORT" || log "sshd not running"
log ""

log "===== LOCAL SSH TEST ====="
python3 - <<'PY' 2>&1 | tee -a "$REPORT"
import socket
try:
    s = socket.create_connection(("127.0.0.1", 8022), timeout=5)
    banner = s.recv(120).decode(errors="ignore").strip()
    print("LOCAL_SSH_8022: PASS")
    print("BANNER:", banner)
    s.close()
except Exception as e:
    print("LOCAL_SSH_8022: FAIL")
    print("ERROR:", e)
PY
log ""

log "===== DELTA MEMORY ====="
MEM="/sdcard/delta_v4_memory.json"

if [ -f "$MEM" ]; then
  ls -lah "$MEM" | tee -a "$REPORT"
  cp "$MEM" "$ROOT/delta_v4_memory_COPY_$TS.json"
  log "Memory copy created in control package."
else
  log "MISSING: $MEM"
fi
log ""

log "===== DELTA JSON VALIDATION ====="
python3 - <<'PY' 2>&1 | tee -a "$REPORT"
import json, os, hashlib
path="/sdcard/delta_v4_memory.json"
if not os.path.exists(path):
    print("JSON_STATUS: MISSING")
else:
    data=open(path,"rb").read()
    print("SIZE_BYTES:", len(data))
    print("SHA256:", hashlib.sha256(data).hexdigest())
    try:
        obj=json.loads(data.decode("utf-8"))
        print("JSON_PARSE: PASS")
        print("ROOT_TYPE:", type(obj).__name__)
        if isinstance(obj, dict):
            print("TOP_KEYS:", list(obj.keys())[:30])
    except Exception as e:
        print("JSON_PARSE: FAIL")
        print("ERROR:", e)
PY
log ""

log "===== DELTA MODULE TEST ====="
python3 - <<'PY' 2>&1 | tee -a "$REPORT"
try:
    from delta_v4_5 import DeltaV45
    d = DeltaV45()
    print("DELTA_V4_5_IMPORT: PASS")
    print("OBJECT:", type(d).__name__)
except Exception as e:
    print("DELTA_V4_5_IMPORT: FAIL")
    print("ERROR:", e)
PY
log ""

log "===== AUTOMATION CONTROL DECISION ====="
log "Old 60-second Lenovo loop remains blocked."
log "No automatic harvest is approved."
log "Android memory is valid only when hash-verified."
log "SSHD is manual transport only."
log "Lenovo sync is allowed only after network port 8022 passes."
log ""

hash_file "$REPORT"
hash_file "$MEM"
if [ -f "$ROOT/delta_v4_memory_COPY_$TS.json" ]; then
  hash_file "$ROOT/delta_v4_memory_COPY_$TS.json"
fi

log ""
log "===== PACKAGE LOCATION ====="
log "$ROOT"
log ""
log "CONTROL STATUS: ANDROID_FULL_CONTROL_PACKAGE_CREATED"

echo ""
echo "ANDROID FULL CONTROL PACKAGE CREATED:"
echo "$ROOT"
echo ""
echo "REPORT:"
echo "$REPORT"
echo ""
echo "HASH REGISTER:"
echo "$HASHREG"
