#!/usr/bin/env python3
"""
VDR Builder v3 (Data Room Engine)
Institutional, evidence-backed generator for PPP / EIB / EBRD style submissions.

Core capabilities
- Loads a JSON project dossier (SSOT)
- Registers annexes/evidence from source files
- Optional light text extraction from PDFs/DOCX/TXT for searchable snippets
- Validates cross-document consistency
- Builds lender-grade DOCX with annex index, evidence map, and validation notes
- Exports annex register and validation report as JSON
- Optional PDF conversion via LibreOffice

No OCR is performed. For scanned PDFs, use manual metadata in the dossier.
"""

# ------------------------- Optional parsers -------------------------

def _safe_import_pypdf():
    try:
        from pypdf import PdfReader
        return PdfReader
    except Exception:
        return None

def _safe_import_docx2txt():
    try:
        import docx2txt
        return docx2txt
    except Exception:
        return None


# ------------------------- Data models -------------------------

@dataclass
class Reference:
    doc_code: str
    description: str
    page_hint: Optional[str] = None

@dataclass
class Annex:
    code: str
    name: str
    path: str
    category: str
    issuer: Optional[str] = None
    issue_date: Optional[str] = None
    evidence_points: List[str] = field(default_factory=list)
    notes: Optional[str] = None
    sha256: Optional[str] = None
    file_size: Optional[int] = None
    page_count: Optional[int] = None
    extracted_snippet: Optional[str] = None
    exists: bool = False

@dataclass
class ValidationItem:
    code: str
    severity: str  # INFO / WARN / ERROR
    message: str
    context: Optional[Dict[str, Any]] = None

@dataclass
class ProjectDossier:
    metadata: Dict[str, Any]
    project_identity: Dict[str, Any]
    project_site: Dict[str, Any]
    financials: Dict[str, Any]
    financing_structure: Dict[str, Any]
    legal_basis: Dict[str, Any]
    esg_alignment: Dict[str, Any]
    narrative: Dict[str, Any]
    signatories: Dict[str, Any]
    annexes: List[Annex]
    evidence_map: Dict[str, List[Reference]]
    assumptions_and_notes: Dict[str, Any] = field(default_factory=dict)


# ------------------------- Utilities -------------------------

def read_json(path: Path) -> Dict[str, Any]:
    with path.open("r", encoding="utf-8") as f:
        return json.load(f)

def write_json(path: Path, data: Any) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("w", encoding="utf-8") as f:
        json.dump(data, f, ensure_ascii=False, indent=2)

def sha256_file(path: Path) -> str:
    h = hashlib.sha256()
    with path.open("rb") as f:
        for chunk in iter(lambda: f.read(1024 * 1024), b""):
            h.update(chunk)
    return h.hexdigest()

def format_currency(value: Any, decimals: int = 0) -> str:
    if value is None or value == "":
        return "—"
    try:
        n = float(value)
    except Exception:
        return str(value)
    if decimals == 0:
        s = f"{n:,.0f}"
    else:
        s = f"{n:,.{decimals}f}"
    return s.replace(",", "X").replace(".", ",").replace("X", ".")

def safe_get(d: Dict[str, Any], *keys: str, default=None):
    cur = d
    for k in keys:
        if not isinstance(cur, dict) or k not in cur:
            return default
        cur = cur[k]
    return cur

def normalize_path(path_str: str, base_dir: Path) -> Path:
    p = Path(path_str)
    return p if p.is_absolute() else (base_dir / p).resolve()

def text_excerpt(text: str, max_len: int = 600) -> str:
    text = re.sub(r"\s+", " ", text or "").strip()
    return text[:max_len] + ("…" if len(text) > max_len else "")


# ------------------------- Extraction -------------------------

def extract_pdf_text(path: Path, max_pages: int = 3) -> Tuple[Optional[str], Optional[int]]:
    PdfReader = _safe_import_pypdf()
    if PdfReader is None:
        return None, None
    try:
        reader = PdfReader(str(path))
        pages = len(reader.pages)
        chunks = []
        for i, page in enumerate(reader.pages[:max_pages]):
            try:
                chunks.append(page.extract_text() or "")
            except Exception:
                continue
        text = "\n".join(chunks).strip() or None
        return text, pages
    except Exception:
        return None, None

def extract_docx_text(path: Path) -> Optional[str]:
    try:
        docx2txt = _safe_import_docx2txt()
        if docx2txt:
            text = docx2txt.process(str(path))
            return text.strip() or None
    except Exception:
        pass
    return None

def extract_text_any(path: Path) -> Tuple[Optional[str], Optional[int]]:
    suffix = path.suffix.lower()
    if suffix == ".pdf":
        return extract_pdf_text(path)
    if suffix == ".docx":
        return extract_docx_text(path), None
    if suffix in {".txt", ".md", ".json"}:
        try:
            return path.read_text(encoding="utf-8", errors="ignore"), None
        except Exception:
            return None, None
    return None, None


# ------------------------- Dossier loading -------------------------

def load_dossier(json_path: Path) -> ProjectDossier:
    raw = read_json(json_path)
    annexes = [Annex(**a) for a in raw.get("annexes", [])]

    evidence_map = {}
    for key, refs in raw.get("evidence_map", {}).items():
        evidence_map[key] = [Reference(**r) for r in refs]

    return ProjectDossier(
        metadata=raw.get("metadata", {}),
        project_identity=raw.get("project_identity", {}),
        project_site=raw.get("project_site", {}),
        financials=raw.get("financials", {}),
        financing_structure=raw.get("financing_structure", {}),
        legal_basis=raw.get("legal_basis", {}),
        esg_alignment=raw.get("esg_alignment", {}),
        narrative=raw.get("narrative", {}),
        signatories=raw.get("signatories", {}),
        annexes=annexes,
        evidence_map=evidence_map,
        assumptions_and_notes=raw.get("assumptions_and_notes", {}),
    )


# ------------------------- Annex registry -------------------------

class AnnexRegistry:
    def __init__(self, base_dir: Path, annexes: List[Annex]):
        self.base_dir = base_dir
        self.annexes = annexes

    def enrich(self) -> None:
        for annex in self.annexes:
            p = normalize_path(annex.path, self.base_dir)
            annex.exists = p.exists()
            if not annex.exists:
                continue
            annex.path = str(p)
            annex.file_size = p.stat().st_size
            annex.sha256 = sha256_file(p)
            text, pages = extract_text_any(p)
            annex.page_count = pages
            annex.extracted_snippet = text_excerpt(text) if text else annex.extracted_snippet

    def to_rows(self) -> List[Dict[str, Any]]:
        return [asdict(a) for a in self.annexes]


# ------------------------- Validation -------------------------

class Validator:
    def __init__(self, dossier: ProjectDossier):
        self.dossier = dossier
        self.items: List[ValidationItem] = []

    def add(self, code: str, severity: str, message: str, context: Optional[Dict[str, Any]] = None) -> None:
        self.items.append(ValidationItem(code=code, severity=severity, message=message, context=context))

    def validate(self) -> List[ValidationItem]:
        self._validate_annexes_exist()
        self._validate_capex_totals()
        self._validate_land_consistency()
        self._validate_legal_identity()
        self._validate_financing()
        self._validate_signatories()
        return self.items

    def _validate_annexes_exist(self):
        missing = [a.code for a in self.dossier.annexes if not a.exists]
        if missing:
            self.add("ANNEX_EXISTS", "ERROR", "One or more annex files are missing.", {"missing": missing})
        else:
            self.add("ANNEX_EXISTS", "INFO", "All annex files resolved successfully.")

    def _validate_capex_totals(self):
        lines = self.dossier.financials.get("capex_lines", [])
        stated_total = self.dossier.financials.get("total_capex_eur")
        calc_total = 0.0
        for x in lines:
            try:
                calc_total += float(x.get("amount_eur", 0))
            except Exception:
                self.add("CAPEX_PARSE", "WARN", "Non-numeric CAPEX line encountered.", {"line": x})
        if stated_total is None:
            self.add("CAPEX_TOTAL", "WARN", "No stated total CAPEX provided.")
            return
        try:
            stated_total = float(stated_total)
        except Exception:
            self.add("CAPEX_TOTAL", "ERROR", "Stated total CAPEX is not numeric.", {"value": stated_total})
            return
        if round(calc_total, 2) != round(stated_total, 2):
            self.add(
                "CAPEX_TOTAL",
                "ERROR",
                "CAPEX line-item total does not reconcile to stated total.",
                {"calculated": calc_total, "stated": stated_total},
            )
        else:
            self.add("CAPEX_TOTAL", "INFO", "CAPEX line-item total reconciles.", {"total": stated_total})

    def _validate_land_consistency(self):
        land_book = safe_get(self.dossier.financials, "land_book_value_eur")
        land_project = safe_get(self.dossier.financials, "land_project_value_eur")
        if land_book is None or land_project is None:
            return
        try:
            book = float(land_book)
            proj = float(land_project)
        except Exception:
            self.add("LAND_VALUE", "WARN", "Land value fields are not numeric.")
            return
        if proj > book * 1.25:
            explanation = safe_get(self.dossier.assumptions_and_notes, "land_value_reconciliation_note")
            sev = "WARN" if explanation else "ERROR"
            msg = "Project land value materially exceeds booked land value."
            if explanation:
                msg += " Reconciliation note present."
            else:
                msg += " No reconciliation note provided."
            self.add("LAND_VALUE", sev, msg, {"book_value": book, "project_value": proj, "multiple": proj / max(book, 1)})

    def _validate_legal_identity(self):
        company = self.dossier.project_identity.get("sponsor_company")
        pib = self.dossier.project_identity.get("pib")
        ubo = self.dossier.project_identity.get("ubo")
        if not company:
            self.add("LEGAL_ID", "ERROR", "Sponsor company is missing.")
        if not pib:
            self.add("LEGAL_ID", "WARN", "PIB is missing.")
        if not ubo:
            self.add("LEGAL_ID", "WARN", "UBO is missing.")

    def _validate_financing(self):
        total = self.dossier.financials.get("total_capex_eur")
        debt = self.dossier.financing_structure.get("debt_eur")
        equity = self.dossier.financing_structure.get("equity_eur")
        if total is None or debt is None or equity is None:
            self.add("FUNDING", "WARN", "Funding structure incomplete.")
            return
        try:
            total = float(total)
            debt = float(debt)
            equity = float(equity)
        except Exception:
            self.add("FUNDING", "ERROR", "Funding structure contains non-numeric values.")
            return
        if round(debt + equity, 2) != round(total, 2):
            self.add("FUNDING", "ERROR", "Debt + Equity does not equal total CAPEX.", {"debt": debt, "equity": equity, "total": total})
        else:
            self.add("FUNDING", "INFO", "Funding structure reconciles to total CAPEX.")
        dsc = self.dossier.financing_structure.get("target_dscr")
        if dsc is not None:
            try:
                dsc = float(dsc)
                if dsc < 1.35:
                    self.add("DSCR", "WARN", "Target DSCR is below lender threshold of 1.35x.", {"target_dscr": dsc})
            except Exception:
                self.add("DSCR", "WARN", "Target DSCR is not numeric.", {"target_dscr": dsc})

    def _validate_signatories(self):
        ed = self.dossier.signatories.get("executive_director")
        ubo = self.dossier.signatories.get("ubo")
        if not ed or not ubo:
            self.add("SIG", "WARN", "One or more signatory blocks are incomplete.")


# ------------------------- DOCX builder -------------------------

class VdrDocBuilder:
    NAVY = RGBColor(31, 78, 121)
    GREEN = RGBColor(40, 167, 69)
    DARK = RGBColor(44, 62, 80)
    GREY = RGBColor(100, 100, 100)

    def __init__(self, dossier: ProjectDossier, validations: List[ValidationItem], out_dir: Path, logo: Optional[Path] = None):
        self.dossier = dossier
        self.validations = validations
        self.out_dir = out_dir
        self.logo = logo
        self.doc = Document()
        self._setup_document()

    def _setup_document(self) -> None:
        sec = self.doc.sections[0]
        sec.page_width = Cm(21.0)
        sec.page_height = Cm(29.7)
        sec.left_margin = Cm(1.6)
        sec.right_margin = Cm(1.6)
        sec.top_margin = Cm(1.2)
        sec.bottom_margin = Cm(1.2)

        self._style("Normal", "Segoe UI", 10, None)
        self._heading_style("Heading 1", 13, self.NAVY)
        self._heading_style("Heading 2", 11, self.DARK)

        if "DocTitle" not in [s.name for s in self.doc.styles]:
            style = self.doc.styles.add_style("DocTitle", WD_STYLE_TYPE.PARAGRAPH)
            style.font.name = "Segoe UI"
            style.font.size = Pt(18)
            style.font.bold = True
            style.font.color.rgb = self.DARK

        self._build_header_footer()

    def _style(self, style_name: str, font_name: str, size_pt: int, color: Optional[RGBColor]):
        style = self.doc.styles[style_name]
        style.font.name = font_name
        style.font.size = Pt(size_pt)
        style._element.rPr.rFonts.set(qn("w:eastAsia"), font_name)
        if color:
            style.font.color.rgb = color

    def _heading_style(self, style_name: str, size_pt: int, color: RGBColor):
        style = self.doc.styles[style_name]
        style.font.name = "Segoe UI Semibold"
        style.font.size = Pt(size_pt)
        style.font.bold = True
        style.font.color.rgb = color
        style._element.rPr.rFonts.set(qn("w:eastAsia"), "Segoe UI Semibold")

    def _shade_cell(self, cell, fill_color: str):
        tcPr = cell._tc.get_or_add_tcPr()
        shd = OxmlElement("w:shd")
        shd.set(qn("w:fill"), fill_color)
        shd.set(qn("w:val"), "clear")
        tcPr.append(shd)

    def _set_borders(self, table, color: str = "1F4E79", size: int = 6):
        tblPr = table._tbl.tblPr
        borders = OxmlElement("w:tblBorders")
        for name in ["top", "left", "bottom", "right", "insideH", "insideV"]:
            border = OxmlElement(f"w:{name}")
            border.set(qn("w:val"), "single")
            border.set(qn("w:sz"), str(size))
            border.set(qn("w:color"), color)
            border.set(qn("w:space"), "0")
            borders.append(border)
        tblPr.append(borders)

    def _add_page_field(self, paragraph):
        run = paragraph.add_run()
        fld1 = OxmlElement("w:fldChar")
        fld1.set(qn("w:fldCharType"), "begin")
        instr = OxmlElement("w:instrText")
        instr.set(qn("xml:space"), "preserve")
        instr.text = " PAGE "
        fld2 = OxmlElement("w:fldChar")
        fld2.set(qn("w:fldCharType"), "end")
        run._r.append(fld1)
        run._r.append(instr)
        run._r.append(fld2)

    def _build_header_footer(self):
        sec = self.doc.sections[0]
        header = sec.header
        t = header.add_table(rows=1, cols=3, width=Cm(18))
        t.autofit = False

        c0, c1, c2 = t.rows[0].cells
        if self.logo and self.logo.exists():
            p = c0.paragraphs[0]
            p.add_run().add_picture(str(self.logo), width=Cm(3.2))
        else:
            p = c0.paragraphs[0]
            p.add_run("NO LOGO").italic = True

        p = c1.paragraphs[0]
        p.alignment = WD_ALIGN_PARAGRAPH.CENTER
        r = p.add_run(self.dossier.metadata.get("institution_name", "MONTENEGRIN INVESTMENT AGENCY"))
        r.bold = True
        r.font.size = Pt(9)
        r.font.color.rgb = self.NAVY
        p.add_run("\n")
        r2 = p.add_run(self.dossier.metadata.get("header_reference", "VDR • TITAN GRID"))
        r2.font.size = Pt(8)
        r2.font.color.rgb = self.GREY

        p = c2.paragraphs[0]
        p.alignment = WD_ALIGN_PARAGRAPH.RIGHT
        r3 = p.add_run(self.dossier.metadata.get("form_reference", "FORM PS-02"))
        r3.font.size = Pt(8)
        r3.font.color.rgb = self.DARK
        p.add_run("\n")
        r4 = p.add_run(self.dossier.metadata.get("system_reference", "CENTRAL BRAIN"))
        r4.font.size = Pt(8)
        r4.font.color.rgb = self.DARK

        footer = sec.footer
        p = footer.paragraphs[0]
        p.alignment = WD_ALIGN_PARAGRAPH.CENTER
        text = self.dossier.metadata.get("footer_text", "CONFIDENTIAL")
        r = p.add_run(text + " • Page ")
        r.font.size = Pt(7)
        r.font.color.rgb = self.GREY
        self._add_page_field(p)

    def build(self) -> Path:
        self._cover()
        self._version_block()
        self._identity()
        self._site()
        self._financials()
        self._financing()
        self._legal_basis()
        self._esg()
        self._narrative()
        self._evidence_map()
        self._annex_index()
        self._validation_section()
        self._signature_section()

        filename = self.dossier.metadata.get("output_filename") or f"MNE-TITAN-VDR-PS02-v3.0-{dt.date.today().isoformat()}.docx"
        out = self.out_dir / filename
        self.out_dir.mkdir(parents=True, exist_ok=True)
        self.doc.save(out)
        return out

    def _cover(self):
        p = self.doc.add_paragraph(self.dossier.metadata.get("title", "STRATEŠKA INVESTICIJSKA REGISTRACIJA"), style="DocTitle")
        p.alignment = WD_ALIGN_PARAGRAPH.CENTER
        p2 = self.doc.add_paragraph(self.dossier.metadata.get("subtitle", "FORM PS-02 | EVIDENCE-BACKED | DATA ROOM ENGINE"))
        p2.alignment = WD_ALIGN_PARAGRAPH.CENTER
        for run in p2.runs:
            run.font.size = Pt(9)
            run.font.color.rgb = self.GREY

        tbl = self.doc.add_table(rows=1, cols=1)
        tbl.alignment = WD_TABLE_ALIGNMENT.CENTER
        self._shade_cell(tbl.rows[0].cells[0], "1F4E79")
        tbl.rows[0].height = Cm(0.12)
        self.doc.add_paragraph()

    def _version_block(self):
        meta = self.dossier.metadata
        p = self.doc.add_paragraph()
        p.alignment = WD_ALIGN_PARAGRAPH.CENTER
        t = (
            f"Version: {meta.get('version','v3.0')} | "
            f"Owner: {meta.get('owner','Central Brain')} | "
            f"Status: {meta.get('status','Draft')} | "
            f"As of: {meta.get('as_of_date', dt.date.today().isoformat())}"
        )
        r = p.add_run(t)
        r.font.size = Pt(8)
        r.font.color.rgb = self.GREY
        self.doc.add_paragraph()

    def _add_kv_table(self, rows: List[Tuple[str, str]], shade_first_col: bool = True):
        t = self.doc.add_table(rows=len(rows), cols=2)
        t.alignment = WD_TABLE_ALIGNMENT.CENTER
        self._set_borders(t, "1F4E79", 5)
        for i, (k, v) in enumerate(rows):
            c0, c1 = t.rows[i].cells
            c0.text = k
            c1.text = v
            if c0.paragraphs[0].runs:
                c0.paragraphs[0].runs[0].bold = True
                c0.paragraphs[0].runs[0].font.size = Pt(9)
                c0.paragraphs[0].runs[0].font.color.rgb = self.NAVY
            if c1.paragraphs[0].runs:
                c1.paragraphs[0].runs[0].font.size = Pt(9)
            if shade_first_col:
                self._shade_cell(c0, "E8F0F8")
        self.doc.add_paragraph()

    def _identity(self):
        d = self.dossier.project_identity
        self.doc.add_heading("1. PROJECT IDENTITY / PROJEKTNI IDENTITET", level=1)
        rows = [
            ("Project name", str(d.get("project_name", "—"))),
            ("Sponsor company", str(d.get("sponsor_company", "—"))),
            ("Category", str(d.get("category", "—"))),
            ("Executive director", str(d.get("executive_director", "—"))),
            ("UBO", str(d.get("ubo", "—"))),
            ("PIB", str(d.get("pib", "—"))),
        ]
        self._add_kv_table(rows)

    def _site(self):
        d = self.dossier.project_site
        self.doc.add_heading("2. PROJECT SITE / LOKACIJA", level=1)
        rows = [
            ("Parcel", str(d.get("parcel", "—"))),
            ("Municipality", str(d.get("municipality", "—"))),
            ("Site area", str(d.get("site_area_m2", "—")) + " m²" if d.get("site_area_m2") else "—"),
            ("Building dimensions", str(d.get("building_dimensions", "—"))),
            ("Use", str(d.get("use", "—"))),
            ("Owner", str(d.get("owner", "—"))),
        ]
        self._add_kv_table(rows)
        refs = self.dossier.evidence_map.get("project_site", [])
        self._add_reference_paragraph(refs)

    def _financials(self):
        d = self.dossier.financials
        self.doc.add_heading("3. FINANCIAL CONSISTENCY / FINANSIJSKA KONZISTENTNOST", level=1)
        intro = self.doc.add_paragraph("Total investment is fixed and presented on a line-item basis with evidence mapping.")
        intro.runs[0].italic = True
        intro.runs[0].font.size = Pt(9)

        lines = d.get("capex_lines", [])
        t = self.doc.add_table(rows=max(2, len(lines) + 2), cols=3)
        t.alignment = WD_TABLE_ALIGNMENT.CENTER
        self._set_borders(t, "1F4E79", 4)
        headers = ["Category", "Amount (€)", "Technical note"]
        for j, h in enumerate(headers):
            cell = t.rows[0].cells[j]
            cell.text = h
            for para in cell.paragraphs:
                para.alignment = WD_ALIGN_PARAGRAPH.CENTER
                if para.runs:
                    para.runs[0].bold = True
                    para.runs[0].font.size = Pt(9)
                    para.runs[0].font.color.rgb = RGBColor(255,255,255)
            self._shade_cell(cell, "1F4E79")

        for i, line in enumerate(lines, start=1):
            t.rows[i].cells[0].text = str(line.get("category", "—"))
            t.rows[i].cells[1].text = format_currency(line.get("amount_eur", "—"))
            t.rows[i].cells[2].text = str(line.get("technical_note", "—"))
            if i % 2 == 1:
                for c in t.rows[i].cells:
                    self._shade_cell(c, "F7FBFD")

        total_row = len(lines) + 1
        t.rows[total_row].cells[0].text = "TOTAL"
        t.rows[total_row].cells[1].text = format_currency(d.get("total_capex_eur"))
        t.rows[total_row].cells[2].text = str(d.get("total_note", "—"))
        for c in t.rows[total_row].cells:
            self._shade_cell(c, "D4EDDA")
            if c.paragraphs[0].runs:
                c.paragraphs[0].runs[0].bold = True

        self.doc.add_paragraph()

        land_book = d.get("land_book_value_eur")
        land_proj = d.get("land_project_value_eur")
        if land_book is not None or land_proj is not None:
            self.doc.add_heading("3.1 Land value bridge", level=2)
            note = (
                f"Booked land value: €{format_currency(land_book)} | "
                f"Project land value: €{format_currency(land_proj)}."
            )
            p = self.doc.add_paragraph(note)
            p.runs[0].font.size = Pt(9)
            recon = self.dossier.assumptions_and_notes.get("land_value_reconciliation_note")
            if recon:
                p2 = self.doc.add_paragraph("Reconciliation note: " + recon)
                p2.runs[0].font.size = Pt(9)

        refs = self.dossier.evidence_map.get("financials", [])
        self._add_reference_paragraph(refs)

    def _financing(self):
        d = self.dossier.financing_structure
        self.doc.add_heading("4. FINANCING STRUCTURE / STRUKTURA FINANSIRANJA", level=1)
        rows = [
            ("Debt (€)", format_currency(d.get("debt_eur"))),
            ("Equity (€)", format_currency(d.get("equity_eur"))),
            ("Debt share", str(d.get("debt_share_pct", "—"))),
            ("Equity share", str(d.get("equity_share_pct", "—"))),
            ("Target DSCR", str(d.get("target_dscr", "—"))),
            ("Grace period", str(d.get("grace_period_months", "—")) + " months" if d.get("grace_period_months") else "—"),
        ]
        self._add_kv_table(rows)
        refs = self.dossier.evidence_map.get("financing_structure", [])
        self._add_reference_paragraph(refs)

    def _legal_basis(self):
        d = self.dossier.legal_basis
        self.doc.add_heading("5. LEGAL BASIS / PRAVNI OSNOV", level=1)
        bullets = d.get("bullets", [])
        for b in bullets:
            self.doc.add_paragraph(str(b), style="List Bullet")
        refs = self.dossier.evidence_map.get("legal_basis", [])
        self._add_reference_paragraph(refs)
        self.doc.add_paragraph()

    def _esg(self):
        d = self.dossier.esg_alignment
        self.doc.add_heading("6. ESG ALIGNMENT", level=1)
        bullets = d.get("bullets", [])
        for b in bullets:
            self.doc.add_paragraph(str(b), style="List Bullet")
        self.doc.add_paragraph()

    def _narrative(self):
        d = self.dossier.narrative
        self.doc.add_heading("7. STRATEGIC RATIONALE / STRATEŠKI RACIONAL", level=1)
        t = self.doc.add_table(rows=1, cols=1)
        t.alignment = WD_TABLE_ALIGNMENT.CENTER
        self._set_borders(t, "28A745", 8)
        c = t.rows[0].cells[0]
        self._shade_cell(c, "E8F5E9")
        p = c.paragraphs[0]
        p.clear()
        r = p.add_run(str(d.get("summary", "—")))
        r.font.size = Pt(10)
        r.bold = True
        r.font.color.rgb = RGBColor(27, 94, 32)
        if d.get("detail"):
            p.add_run(" " + str(d["detail"])).font.size = Pt(10)
        self.doc.add_paragraph()

    def _add_reference_paragraph(self, refs: List[Reference]):
        if not refs:
            return
        p = self.doc.add_paragraph()
        p.add_run("Evidence references: ").bold = True
        for idx, ref in enumerate(refs):
            frag = f"{ref.doc_code} ({ref.description}"
            if ref.page_hint:
                frag += f", {ref.page_hint}"
            frag += ")"
            p.add_run(frag)
            if idx < len(refs) - 1:
                p.add_run("; ")
        for r in p.runs:
            r.font.size = Pt(8)

    def _evidence_map(self):
        self.doc.add_heading("8. EVIDENCE MAP / MAPA DOKAZA", level=1)
        t = self.doc.add_table(rows=1, cols=3)
        t.alignment = WD_TABLE_ALIGNMENT.CENTER
        self._set_borders(t, "1F4E79", 4)
        for j, h in enumerate(["Statement", "Supporting documents", "Comment"]):
            t.rows[0].cells[j].text = h
            self._shade_cell(t.rows[0].cells[j], "1F4E79")
            if t.rows[0].cells[j].paragraphs[0].runs:
                run = t.rows[0].cells[j].paragraphs[0].runs[0]
                run.bold = True
                run.font.color.rgb = RGBColor(255,255,255)

        for key, refs in self.dossier.evidence_map.items():
            row = t.add_row()
            row.cells[0].text = key
            row.cells[1].text = "; ".join([f"{r.doc_code} - {r.description}" for r in refs])
            row.cells[2].text = "Mapped"
        self.doc.add_paragraph()

    def _annex_index(self):
        self.doc.add_heading("9. ANNEX INDEX / REGISTAR ANEKSA", level=1)
        t = self.doc.add_table(rows=1, cols=6)
        t.alignment = WD_TABLE_ALIGNMENT.CENTER

        headers = ["Code", "Name", "Category", "Issue date", "Pages", "File"]
        for j, h in enumerate(headers):

                run = t.rows[0].cells[j].paragraphs[0].runs[0]
                run.bold = True
                run.font.color.rgb = RGBColor(255,255,255)

        for i, annex in enumerate(self.dossier.annexes, start=1):
            row = t.add_row()
            row.cells[0].text = annex.code
            row.cells[1].text = annex.name
            row.cells[2].text = annex.category
            row.cells[3].text = annex.issue_date or "—"
            row.cells[4].text = str(annex.page_count or "—")
            row.cells[5].text = Path(annex.path).name if annex.path else "—"
            if i % 2 == 1:
                for c in row.cells:
                    self._shade_cell(c, "F7FBFD")
        self.doc.add_paragraph()

    def _validation_section(self):
        self.doc.add_heading("10. VALIDATION REPORT / IZVJEŠTAJ O VALIDACIJI", level=1)
        t = self.doc.add_table(rows=1, cols=4)
        t.alignment = WD_TABLE_ALIGNMENT.CENTER
        self._set_borders(t, "1F4E79", 4)
        for j, h in enumerate(["Code", "Severity", "Message", "Context"]):
            t.rows[0].cells[j].text = h
            self._shade_cell(t.rows[0].cells[j], "1F4E79")
            if t.rows[0].cells[j].paragraphs[0].runs:
                run = t.rows[0].cells[j].paragraphs[0].runs[0]
                run.bold = True
                run.font.color.rgb = RGBColor(255,255,255)

        for item in self.validations:
            row = t.add_row()
            row.cells[0].text = item.code
            row.cells[1].text = item.severity
            row.cells[2].text = item.message
            row.cells[3].text = json.dumps(item.context or {}, ensure_ascii=False)
            if item.severity == "ERROR":
                for c in row.cells:
                    self._shade_cell(c, "F8D7DA")
            elif item.severity == "WARN":
                for c in row.cells:
                    self._shade_cell(c, "FFF3CD")
        self.doc.add_paragraph()

    def _signature_section(self):
        s = self.dossier.signatories
        self.doc.add_heading("11. CERTIFICATION & SIGNATURES / CERTIFIKACIJA", level=1)

        p.alignment = WD_ALIGN_PARAGRAPH.CENTER

        t.rows[1].cells[0].text = "\n\n_______________________________\n(signature)"

def build_reports(dossier_path: Path, output_dir: Path, logo: Optional[Path] = None, convert_pdf: bool = False) -> Dict[str, str]:
    dossier = load_dossier(dossier_path)
    registry = AnnexRegistry(dossier_path.parent, dossier.annexes)
    registry.enrich()

    validator = Validator(dossier)
    validations = validator.validate()

    builder = VdrDocBuilder(dossier, validations, output_dir, logo=logo)
    docx_path = builder.build()

    annex_json = output_dir / "annex_register.json"
    validation_json = output_dir / "validation_report.json"
    write_json(annex_json, registry.to_rows())
    write_json(validation_json, [asdict(v) for v in validations])

    pdf_path = None
    if convert_pdf:
        pdf_path = convert_docx_to_pdf(docx_path, output_dir)

    return {
        "docx": str(docx_path),
        "pdf": str(pdf_path) if pdf_path else "",
        "annex_register": str(annex_json),
        "validation_report": str(validation_json),
    }

def convert_docx_to_pdf(docx_path: Path, output_dir: Path) -> Optional[Path]:
    soffice = shutil.which("libreoffice") or shutil.which("soffice")
    if not soffice:
        return None
    try:
        subprocess.run([soffice, "--headless", "--convert-to", "pdf", "--outdir", str(output_dir), str(docx_path)], check=True, capture_output=True)
        return output_dir / (docx_path.stem + ".pdf")
    except Exception:
        return None

def cli():
    parser = argparse.ArgumentParser(description="VDR Builder v3 (Data Room Engine)")
    sub = parser.add_subparsers(dest="cmd")

    p_build = sub.add_parser("build", help="Build DOCX and JSON reports from a dossier")
    p_build.add_argument("--dossier", required=True, help="Path to JSON dossier")
    p_build.add_argument("--output-dir", required=True, help="Output directory")
    p_build.add_argument("--logo", default="", help="Optional logo image path")
    p_build.add_argument("--pdf", action="store_true", help="Attempt PDF conversion")

    p_scaffold = sub.add_parser("scaffold", help="Create an empty sample dossier JSON")
    p_scaffold.add_argument("--output", required=True, help="Output JSON path")

    args = parser.parse_args()

    if args.cmd == "build":
        result = build_reports(
            dossier_path=Path(args.dossier),
            output_dir=Path(args.output_dir),
            logo=Path(args.logo) if args.logo else None,
            convert_pdf=args.pdf,
        )
        print(json.dumps(result, indent=2, ensure_ascii=False))
        return
    if args.cmd == "scaffold":
        scaffold_dossier(Path(args.output))
        print(f"Scaffold created: {args.output}")
        return

    parser.print_help()

if __name__ == "__main__":
    cli()
