#!/usr/bin/env python3
import json
from datetime import datetime
from pathlib import Path

MEMORY_FILE = Path("/sdcard/delta_v4_memory.json")

class DeltaV4:
    def __init__(self):
        self.memory = self._load()
        print("Delta v4.0 pokrenuta.")

    def _load(self):
        if MEMORY_FILE.exists():
            try:
                return json.load(open(MEMORY_FILE))
            except:
                pass
        return {"signals": [], "goals": []}

    def save(self):
        with open(MEMORY_FILE, "w") as f:
            json.dump(self.memory, f, indent=2)

    def ingest(self, text):
        self.memory["signals"].append({"time": datetime.now().isoformat(), "text": text[:200]})
        self.save()
        print("Usvojeno.")

    def add_goal(self, goal):
        self.memory["goals"].append(goal)
        self.save()
        print("Cilj dodat.")

    def reflect(self):
        print("Refleksija zavrsena.")

    def query(self, question):
        q = question.lower()
        results = []
        if "cilj" in q:
            results = self.memory.get("goals", [])[-5:]
        else:
            for s in self.memory.get("signals", []):
                if any(w in str(s).lower() for w in q.split()):
                    results.append(str(s)[:120])
        print("\nPitanje:", question)
        if results:
            for r in results[:5]:
                print("-", r)
        else:
            print("Nemam informacije.")
        print()

    def status(self):
        print("\n=== DELTA v4.0 ===")
        print("Signali:", len(self.memory.get("signals", [])))
        print("Ciljevi:", len(self.memory.get("goals", [])))
        print("==================\n")

if __name__ == "__main__":
    d = DeltaV4()
    d.status()
    d.reflect()
    print("Spremna. Koristi d.query('Koji su moji ciljevi?')")
