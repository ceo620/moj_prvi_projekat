#!/data/data/com.termux/files/usr/bin/env python3
"""
DELTA ORACLE ONESHOT ADAPTER
Purpose:
- Replacement for DELTA_ORACLE.py loop behavior.
- One command, one response, no while True, no sleep loop.
- Uses DeltaV45 only.
"""

import sys
from delta_v4_5 import DeltaV45

def main():
    d = DeltaV45()

    if len(sys.argv) < 2:
        print("Usage:")
        print("  python3 ~/delta_oracle_oneshot_adapter.py status")
        print("  python3 ~/delta_oracle_oneshot_adapter.py query 'question text'")
        return 0

    cmd = sys.argv[1].lower()

    if cmd == "status":
        if hasattr(d, "status"):
            print(d.status())
        else:
            mem = getattr(d, "memory", {})
            print("DELTA ORACLE ONESHOT STATUS")
            print("signals:", len(mem.get("signals", [])))
            print("goals:", len(mem.get("goals", [])))
            print("chat_history:", len(mem.get("chat_history", [])))
            print("facts:", len(mem.get("facts", [])))
            print("reflections:", len(mem.get("reflections", [])))
        return 0

    if cmd == "query":
        question = " ".join(sys.argv[2:]).strip()
        if not question:
            print("ERROR: missing query text")
            return 2

        if hasattr(d, "query"):
            print(d.query(question))
        else:
            print("DELTA QUERY FALLBACK")
            print("Question:", question)
            print("Memory keys:", list(getattr(d, "memory", {}).keys()))
        return 0

    print("ERROR: unknown command:", cmd)
    return 2

if __name__ == "__main__":
    raise SystemExit(main())
