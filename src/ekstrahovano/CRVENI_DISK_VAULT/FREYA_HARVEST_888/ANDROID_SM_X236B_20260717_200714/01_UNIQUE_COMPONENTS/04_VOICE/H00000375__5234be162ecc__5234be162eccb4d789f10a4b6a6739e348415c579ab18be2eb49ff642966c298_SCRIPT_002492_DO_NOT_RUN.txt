#!/bin/zsh

VAULT="/Volumes/DANIJELA HARD DISK/MEC_CMU_MACBOOK_BACKUP_VAULT"
ROOT="$VAULT/27_LENOVO_LINKEDIN_PIPELINE_INTAKE"
DRAFTS="$ROOT/03_AUTOMATED_HOOK_DRAFTS"
CONTROL="$ROOT/99_CONTROL"
STATE="$CONTROL/THREE_DAY_HOOK_DRAFT_STATE.txt"

mkdir -p "$DRAFTS"
mkdir -p "$CONTROL"

STAMP=$(date +"%Y%m%d_%H%M%S")
TODAY=$(date +"%Y-%m-%d")

if [ ! -f "$STATE" ]; then
  echo "0" > "$STATE"
fi

IDX=$(cat "$STATE" 2>/dev/null)
case "$IDX" in
  ''|*[!0-9]*) IDX=0 ;;
esac

NEXT_IDX=$(( (IDX + 1) % 7 ))
echo "$NEXT_IDX" > "$STATE"

case "$IDX" in
  0)
    TITLE="Evidence before claims"
    HOOK="The strongest project is not the loudest one. It is the one whose evidence can survive review."
    BODY="This is the discipline behind TITAN GRID: before a claim becomes public, it must pass through structure, evidence, and human responsibility."
    ;;
  1)
    TITLE="Human-controlled AI"
    HOOK="AI should not replace responsibility. It should protect the human decision before the system moves."
    BODY="TITAN GRID is built around this rule: AI supports the work, but the human stays in control."
    ;;
  2)
    TITLE="Data-room discipline"
    HOOK="A serious project is not only built in documents. It is built in traceability."
    BODY="TITAN GRID organizes evidence, hashes records, controls claims, and prepares project logic for serious review."
    ;;
  3)
    TITLE="Transformer-industry vision"
    HOOK="Transformers are not only machines. They are part of the energy backbone of every serious economy."
    BODY="Through ARS project development, TITAN GRID connects industrial vision with documentation, governance, and evidence-first preparation."
    ;;
  4)
    TITLE="Montenegro to EU, Turkey, and the world"
    HOOK="A system can be built from a small country and still speak the language of global review."
    BODY="TITAN GRID is being developed from Montenegro with respect for European standards, Turkish industrial know-how, and global institutional discipline."
    ;;
  5)
    TITLE="Women in business / founder discipline"
    HOOK="Founders do not only build companies. Sometimes they build the system they needed to survive the pressure."
    BODY="TITAN GRID was born from pressure, mistakes, evidence, and the need to protect every serious step before public claims are made."
    ;;
  6)
    TITLE="What TITAN GRID does now"
    HOOK="TITAN GRID turns scattered work into a controlled evidence chain."
    BODY="It maps risks, protects documents, structures metadata, controls public language, and keeps publication under human review."
    ;;
esac

DRAFT="$DRAFTS/HOOK_DRAFT_${STAMP}.txt"
HASH="$DRAFTS/HOOK_DRAFT_SHA256_${STAMP}.csv"
RUN_STATUS="$DRAFTS/HOOK_DRAFT_STATUS_${STAMP}.txt"

cat > "$DRAFT" <<EOD
LINKEDIN_HOOK_DRAFT
Date: $TODAY
Rotation_Index: $IDX
Theme: $TITLE

HOOK:
$HOOK

BODY:
$BODY

SAFE CTA:
If this resonates with your work in governance, industrial development, responsible AI, or evidence-first project preparation, follow this journey.

SIGNATURE:
Danijela Keskin
Operator & Architect — TITAN GRID
CEO, Ars Metal Industries DOO Montenegro

HASHTAGS:
#TITANGRID #Protocol888 #EvidenceFirst #AIGovernance #DataRoom #HumanControlledAI #Montenegro #CrnaGora #MNE #EuropeanUnion #EU #Turkey #Türkiye #IndustrialDevelopment #ResponsibleAI #WomenInBusiness #BuiltFromMontenegro #GlobalVision #World

CLAIM_CONTROL:
APPROVAL_CLAIM=NO
FUNDING_CLAIM=NO
CONTRACT_CLAIM=NO
PARTNERSHIP_OVERCLAIM=NO
INSTITUTIONAL_VALIDATION_CLAIM=NO
PUBLICATION_MODE=MANUAL_ONLY

GUARDRAILS:
DRAFT_ONLY=YES
AUTO_POSTING_DONE=NO
BROWSER_AUTOMATION_DONE=NO
SCRAPING_DONE=NO
FOLLOWING_DONE=NO
COMMENTING_DONE=NO
DM_DONE=NO
EOD

SHA=$(shasum -a 256 "$DRAFT" | awk '{print $1}')
SIZE=$(stat -f%z "$DRAFT" 2>/dev/null)

{
  echo "SHA256,SIZE_BYTES,FILE_PATH"
  echo "$SHA,$SIZE,\"$DRAFT\""
} > "$HASH"

{
  echo "THREE_DAY_HOOK_DRAFT_STATUS"
  echo "Timestamp: $(date)"
  echo ""
  echo "STATUS:"
  echo "HOOK_DRAFT_CREATED=YES"
  echo ""
  echo "DRAFT=$DRAFT"
  echo "HASH_REGISTER=$HASH"
  echo "SHA256=$SHA"
  echo "SIZE_BYTES=$SIZE"
  echo "ROTATION_INDEX=$IDX"
  echo "NEXT_ROTATION_INDEX=$NEXT_IDX"
  echo ""
  echo "GUARDRAILS:"
  echo "DRAFT_ONLY=YES"
  echo "AUTO_POSTING_DONE=NO"
  echo "BROWSER_AUTOMATION_DONE=NO"
  echo "SCRAPING_DONE=NO"
  echo "FOLLOWING_DONE=NO"
  echo "COMMENTING_DONE=NO"
  echo "DM_DONE=NO"
  echo "DELETE_DONE=NO"
  echo "RUN_ALL_DONE=NO"
  echo ""
  echo "NEXT_ALLOWED:"
  echo "HUMAN_REVIEW_DRAFT"
  echo "MANUAL_PUBLICATION_ONLY"
} > "$RUN_STATUS"

exit 0
