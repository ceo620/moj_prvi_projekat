#!/usr/bin/env python3
"""
DELTA v4.0 — Core Learning Engine + Query
"""

import json
from datetime import datetime
from pathlib import Path

MEMORY_FILE = Path("/sdcard/delta_v4_memory.json")

class DeltaV4:
    def __init__(self):
        self.memory = self._load()
        print("🧠 Delta v4.0 pokrenuta.")

    def _load(self):
        if MEMORY_FILE.exists():
            try:
                return json.load(open(MEMORY_FILE, encoding="utf-8"))
            except:
                pass
        return {
            "created": datetime.now().isoformat(),
            "updated": datetime.now().isoformat(),
            "signals": [],
            "facts": [],
            "goals": [],
            "reflections": [],
            "chat_history": []
        }

    def save(self):
        self.memory["updated"] = datetime.now().isoformat()
        with open(MEMORY_FILE, "w", encoding="utf-8") as f:
            json.dump(self.memory, f, indent=2, ensure_ascii=False)

    def ingest(self, text: str, source: str = "chat"):
        self.memory["chat_history"].append({
            "time": datetime.now().isoformat(),
            "source": source,
            "content": text
        })
        self.memory["signals"].append({
            "time": datetime.now().isoformat(),
            "text": text[:200]
        })
        self.save()
        print(f"📥 Usvojeno: {text[:60]}...")

    def add_goal(self, goal: str):
        self.memory["goals"].append({
            "time": datetime.now().isoformat(),
            "goal": goal
        })
        self.save()
        print(f"🎯 Goal added: {goal}")

    def reflect(self):
        reflection = {
            "time": datetime.now().isoformat(),
            "signals": len(self.memory["signals"]),
            "facts": len(self.memory["facts"]),
            "goals": len(self.memory["goals"]),
            "note": "Delta uči i raste."
        }
        self.memory["reflections"].append(reflection)
        self.save()
        print("🪞 Refleksija završena.")

    def query(self, question: str):
        q = question.lower()
        results = []
        if "cilj" in q or "goal" in q:
            results = [g["goal"] for g in self.memory.get("goals", [])[-5:]]
        elif "fact" in q or "činjenic" in q or "znaš" in q:
            results = [f["fact"] for f in self.memory.get("facts", [])[-8:]]
        else:
            for item in self.memory.get("facts", []) + self.memory.get("signals", []):
                if any(word in str(item).lower() for word in q.split()):
                    results.append(str(item)[:150])
        print(f"\n❓ Pitanje: {question}")
        if results:
            print("📌 Odgovor iz memorije:")
            for r in results[:5]:
                print(f"  - {r}")
        else:
            print("Nemam još dovoljno informacija o tome.")
        print()

    def status(self):
        print("\n" + "="*50)
        print("DELTA v4.0 STATUS")
        print(f"Signali: {len(self.memory['signals'])}")
        print(f"Činjenice: {len(self.memory['facts'])}")
        print(f"Ciljevi: {len(self.memory['goals'])}")
        print("="*50 + "\n")

if __name__ == "__main__":
    d = DeltaV4()
    d.status()
    d.reflect()
    print("Delta v4.0 spremna. Koristi d.query('pitanje') da je pitaš.")
